# 二叉树（Binary Tree）

## 一句话
二叉树是每个节点最多有两个子树（左子树和右子树）的树结构。

## 严格定义
二叉树 $T$ 是一个有限的节点集合，该集合或者为空，或者由一个根节点以及两棵互不相交的、分别称为左子树 $T_L$ 和右子树 $T_R$ 的二叉树组成。

## 遍历（Traversal）
遍历是按特定规则访问树中所有节点各一次的操作。

### 1. 前序遍历 (Preorder Traversal)
**顺序**：根节点 $\to$ 左子树 $\to$ 右子树。

**迭代实现（显式栈）**：
```cpp
vector<int> preorderTraversal(TreeNode* root) {
    vector<int> res;
    if (root == nullptr) return res;
    stack<TreeNode*> stk;
    stk.push(root);
    while (!stk.empty()) {
        TreeNode* node = stk.top();
        stk.pop();
        res.push_back(node->val);
        if (node->right) stk.push(node->right); // 先压右，后出左
        if (node->left) stk.push(node->left);
    }
    return res;
}
```
*注：另一种模拟递归堆栈的写法（模板式）：*
```cpp
vector<int> preorderTraversal(TreeNode* root) {
    vector<int> res;
    stack<TreeNode*> stk;
    TreeNode* node = root;
    while (node != nullptr || !stk.empty()) {
        while (node != nullptr) {
            res.push_back(node->val); // 访问
            stk.push(node);
            node = node->left;
        }
        node = stk.top(); stk.pop();
        node = node->right;
    }
    return res;
}
```

### 2. 中序遍历 (Inorder Traversal)
**顺序**：左子树 $\to$ 根节点 $\to$ 右子树。

**迭代实现（显式栈）**：
```cpp
vector<int> inorderTraversal(TreeNode* root) {
    vector<int> res;
    stack<TreeNode*> stk;
    TreeNode* node = root;
    while (node != nullptr || !stk.empty()) {
        while (node != nullptr) {
            stk.push(node);
            node = node->left;
        }
        node = stk.top(); stk.pop();
        res.push_back(node->val); // 访问
        node = node->right;
    }
    return res;
}
```

### 3. 后序遍历 (Postorder Traversal)
**顺序**：左子树 $\to$ 右子树 $\to$ 根节点。

**迭代实现（显式栈）**：
*技巧：前序是“根左右”，调整为“根右左”，最后反转即为“左右根”。*
```cpp
vector<int> postorderTraversal(TreeNode* root) {
    vector<int> res;
    if (root == nullptr) return res;
    stack<TreeNode*> stk;
    stk.push(root);
    while (!stk.empty()) {
        TreeNode* node = stk.top();
        stk.pop();
        res.push_back(node->val);
        if (node->left) stk.push(node->left);   // 先压左，后出右
        if (node->right) stk.push(node->right);
    }
    reverse(res.begin(), res.end());
    return res;
}
```
*通用迭代写法（防止反转）：*
```cpp
vector<int> postorderTraversal(TreeNode* root) {
    vector<int> res;
    stack<TreeNode*> stk;
    TreeNode *prev = nullptr;
    TreeNode *node = root;
    while (node != nullptr || !stk.empty()) {
        while (node != nullptr) {
            stk.push(node);
            node = node->left;
        }
        node = stk.top(); stk.pop();
        if (node->right == nullptr || node->right == prev) {
            res.push_back(node->val); // 访问
            prev = node;
            node = nullptr; // 强制退栈
        } else {
            stk.push(node); // 再次入栈
            node = node->right;
        }
    }
    return res;
}
```

# 跳表（Skip List）

## 一句话
跳表是在有序链表基础上添加多级索引，实现 O(log n) 查询、插入、删除的概率性数据结构。

## 核心思想
通过随机层数建立多级索引，查询时从高层开始逐层向下缩小范围。

## 完整实现

```cpp
#include <bits/stdc++.h>
using namespace std;

template<typename T, typename Compare = less<T>>
class SkipList {
    struct Node {
        T value;
        vector<Node*> next;
        Node(T v, int level) : value(v), next(level, nullptr) {}
    };

    const int MAX_LEVEL = 16;
    const float P = 0.5;
    Compare cmp;
    Node* head;
    int level;
    mt19937 rng{random_device{}()};
    
    int randomLevel() {
        int lvl = 1;
        while (uniform_real_distribution<float>(0, 1)(rng) < P && lvl < MAX_LEVEL)
            ++lvl;
        return lvl;
    }

public:
    SkipList() : level(1) {
        head = new Node(T{}, MAX_LEVEL);
    }
    
    ~SkipList() {
        Node* cur = head;
        while (cur) {
            Node* nxt = cur->next[0];
            delete cur;
            cur = nxt;
        }
    }

    bool search(T target, bool printPath = false) {
        Node* cur = head;
        vector<pair<int, T>> path;
        
        for (int i = level - 1; i >= 0; --i) {
            while (cur->next[i] && cmp(cur->next[i]->value, target)) {
                cur = cur->next[i];
            }
            if (printPath) path.push_back({i, cur->value});
        }
        
        cur = cur->next[0];
        bool found = (cur && !cmp(cur->value, target) && !cmp(target, cur->value));
        
        if (printPath) {
            cout << "查询 " << target << " 的路径：\n";
            for (auto& [lv, val] : path) 
                cout << "  Level " << lv << ": 停在 " << val << "\n";
            cout << "  Level 0: 检查 " << (cur ? to_string(cur->value) : "nullptr") 
                 << " -> " << (found ? "命中" : "未命中") << "\n";
        }
        return found;
    }

    void insert(T val) {
        vector<Node*> update(MAX_LEVEL, nullptr);
        Node* cur = head;
        
        for (int i = level - 1; i >= 0; --i) {
            while (cur->next[i] && cmp(cur->next[i]->value, val))
                cur = cur->next[i];
            update[i] = cur;
        }
        
        cur = cur->next[0];
        if (!cur || cmp(val, cur->value) || cmp(cur->value, val)) {
            int newLevel = randomLevel();
            if (newLevel > level) {
                for (int i = level; i < newLevel; ++i)
                    update[i] = head;
                level = newLevel;
            }
            
            Node* newNode = new Node(val, newLevel);
            for (int i = 0; i < newLevel; ++i) {
                newNode->next[i] = update[i]->next[i];
                update[i]->next[i] = newNode;
            }
        }
    }

    void display() {
        cout << "\n=== 跳表结构 ===\n";
        for (int i = level - 1; i >= 0; --i) {
            cout << "Level " << i << ": ";
            Node* cur = head->next[i];
            while (cur) {
                cout << cur->value << " -> ";
                cur = cur->next[i];
            }
            cout << "null\n";
        }
        cout << "================\n";
    }
};

int main() {
    SkipList<int> sl;
    vector<int> data = {3, 6, 7, 9, 12, 17, 19, 21, 25, 26};
    for (int x : data) sl.insert(x);
    
    sl.display();
    
    cout << "\n[查询 21]:\n";
    bool r1 = sl.search(21, true);
    cout << "结果: " << (r1 ? "存在" : "不存在") << "\n";
    
    cout << "\n[查询 15]:\n";
    bool r2 = sl.search(15, true);
    cout << "结果: " << (r2 ? "存在" : "不存在") << "\n";
    
    return 0;
}
```

## 复杂度
- **时间**: 平均 O(log n)，最坏 O(n)
- **空间**: 平均 O(n)，每个节点平均 2 层
